# we want to allow 10000 threads in total
# 10000 / 25 processes = 400
<IfModule mpm_event_module>
	ServerLimit 25
	StartServers 2
	MaxRequestWorkers 10000
	MinSpareThreads 401
	MaxSpareThreads 800
	ThreadLimit 400
	ThreadsPerChild 400
	KeepAlive On
	KeepAliveTimeout 5
	MaxKeepAliveRequests 500
	MaxConnectionsPerChild 10000000
</IfModule>


#
# php-fpm config
#

# install.sh sets these in /etc/php/7.4/fpm/pool.d/www.conf (and net.core.somaxconn = 4096 in
# /etc/sysctl.d/60-fpm.conf). On a server installed before 2026-10 set them by hand, then
# `sysctl --system` and `php-fpm7.4 -t && systemctl restart php7.4-fpm` (a reload keeps the old
# listening socket, so listen.backlog only changes on a restart):
#
#   listen.backlog = 1024
#   pm.max_children = 64             ; was 10: one request per child, ~20 MB each
#   pm.start_servers = 16
#   pm.min_spare_servers = 8
#   pm.max_spare_servers = 32
#   pm.max_requests = 10000
#   pm.status_path = /fpm-status     ; served to localhost by the -ssl vhost
#   slowlog = /var/log/php7.4-fpm.slow.log
#   request_slowlog_timeout = 2s     ; backtrace of any request over 2 s
#   request_terminate_timeout = 3600s ; counts from the first byte, and Apache streams the upload body
#                                     ; to php-fpm as it arrives, so it must outlast a 525M upload on a
#                                     ; slow connection (300s = 14 Mbit/s minimum); only a stuck request hits it
#
# Why 64: the hub polls syncClientData once a second with a 1 s timeout, so a stall of a few seconds
# fills a 10-child pool with queued heartbeats and the hub marks the server unreachable. Each uploader
# also costs ~2 upload-progress polls a second plus one child for a few seconds at the end.
# Check, on the box: curl -s "https://$DTCDN_HOSTNAME/fpm-status?full" | head -12
# (the vhost is bound to the host's address, so a request to 127.0.0.1 doesn't reach it; the block's
# `Require local` accepts a connection whose client and server address are the same, which that is)
# (listen queue should be 0, max children reached should stay 0); saturation also logs
# "server reached pm.max_children" in /var/log/php7.4-fpm.log.

# Updating a server that was installed before 2026-10 (the install scripts do all of this on a fresh box)
#

# 1. The pull itself. install.sh ran chmod +x on the scripts, which git shows as mode-only changes (" M"), and
#    the 2026-10 commit sets those same modes, so the pull refuses to overwrite them: put the modes back first.
cd /home/dtcdn
git status --short            # expect only mode changes under daemon/, scripts/, install/, devel/cpp-redis-pipe/build.sh
git checkout -- daemon scripts install devel/cpp-redis-pipe/build.sh
git pull

# 2. The files the pull can't install itself. Sudoers first: until the new rule is in, the minute sweep can't read
#    container states and logs transcode_sweep_unknown_state every minute (it leaves such jobs alone).
cd /home/dtcdn/install
visudo -cf dtcdn-sudoers && sudo install -m 0440 dtcdn-sudoers /etc/sudoers.d/dtcdn-sudoers   # adds scripts/docker-container.sh
sudo cp dtcdn-cron /etc/cron.d/                                                               # output now goes to logs/cron-*.log
sudo cp dtcdn-logrotate /etc/logrotate.d/dtcdn
sudo cp dtcdn-docker-events.service /etc/systemd/system/ && sudo systemctl daemon-reload && sudo systemctl restart dtcdn-docker-events
systemctl status dtcdn-docker-events --no-pager | head -5
